
INSTALACIÓN DE DOCKER EN KALI LINUX
https://hackunderway.com/instalacion-de-docker-en-kali-linux/

*** 1 actualizar lista de paquetes
sudo apt update && sudo apt upgrade -y

*** 2 instalar dependenciias necesarias
sudo apt install apt-transport-https ca-certificates curl gnupg lsb-release -y

*** 3 instala clave GPG oficial de Docker
curl -fsSL https://download.docker.com/linux/debian/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg

*** 4 Agregue el repositorio de Docker a las fuentes de APT:
echo \
"deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] \
https://download.docker.com/linux/debian bullseye stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

*** 5 instalar docker engine
sudo apt update
sudo apt install docker-ce docker-ce-cli containerd.io -y

*** 6 habilita e inicia el servicio de Docker
sudo systemctl enable docker
sudo systemctl start docker

*** 7 verifica que funciona
sudo docker run hello-world

--- para ejecutar docker sin sudo
sudo usermod -aG docker $USER
